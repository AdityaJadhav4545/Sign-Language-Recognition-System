# Real-time inference loop
while True:
    ok, frame = cap.read()
    frame = cv2.flip(frame, 1)

    # 1. MediaPipe landmark detection
    rgb  = cv2.cvtColor(frame, BGR2RGB)
    res  = hands.process(rgb)

    # 2. Normalize landmarks → (63,)
    lm   = extract_landmarks(res)
    feat = normalize_landmarks(lm)

    # 3. ONNX inference → <3ms
    logits = session.run(None,
        {"landmarks": feat[None]})[0]
    probs  = softmax(logits[0])
    sign   = classes[probs.argmax()]

    # 4. Debounce → phrase
    phrase_buf.push(sign)
    phrase = phrase_buf.get_phrase()

    # 5. Render HUD
    draw_hud(frame, sign, phrase)
    cv2.imshow("SignSense", frame)

$ python src/train.py --config configs/static.yaml --model static # train (~5 min)
$ python src/inference.py --config configs/static.yaml # launch webcam
